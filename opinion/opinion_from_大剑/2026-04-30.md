def retry_on_error(max_retries=3, delay=2, exceptions=(ConnectionError, TimeoutError)):
    def decorator(fn):
        @wraps(fn)
        def wrapper(*args, **kwargs):
            for attempt in range(max_retries):
                try:
                    return fn(*args, **kwargs)
                except exceptions as e:
                    if attempt == max_retries - 1:
                        raise
                    time.sleep(delay * (2 ** attempt))
                except Exception:
                    raise  # 非网络错误直接抛出
            return None
        return wrapper
    return decorator
```

- **建议修复**：
```python
def retry_on_error(max_retries=3, delay=2, exceptions=(ConnectionError, TimeoutError)):
    def decorator(fn):
        @wraps(fn)
        def wrapper(*args, **kwargs):
            for attempt in range(max_retries):
                try:
                    return fn(*args, **kwargs)
                except exceptions as e:
                    if attempt == max_retries - 1:
                        raise
                    time.sleep(delay * (2 ** attempt))
                except Exception:
                    raise  # 非网络错误直接抛出
            return None
        return wrapper
    return decorator
```
