```python
def chinese_to_arabic(s: str) -> str:
    if s.isdigit():
        return s
    units = {'十': 10, '百': 100, '千': 1000}
    nums = {'零': 0, '一': 1, '二': 2, '三': 3, '四': 4, '五': 5,
            '六': 6, '七': 7, '八': 8, '九': 9}
    result = 0
    current = 0
    for char in s:
        if char in nums:
            current = nums[char]
        elif char in units:
            if current == 0:
                current = 1
            result += current * units[char]
            current = 0
    result += current
    return str(result) if result > 0 else ""
```
