```patch
def append_reply(filepath, content):
    # 清理任何残留的冲突标记
    content = re.sub(r'^=======\s*$', '', content, flags=re.MULTILINE)
    separator = '\n---\n'
    with open(filepath, 'r+') as f:
        # 检查文件末尾是否已存在分隔符
        try:
            f.seek(-len(separator), 2)
            if f.read(len(separator)) == separator:
                # 已有分隔符，直接追加内容
                f.write(content)
                return
        except OSError:
            pass
        # 文件为空或无分隔符，正常追加
        f.write(separator + content)
```
